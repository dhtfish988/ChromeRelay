cmake_minimum_required(VERSION 3.24)
project(ChromeRelay VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CHROMERELAY_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(CHROMERELAY_BUILD_FUZZER "Build bounded protocol fuzz target with LLVM libFuzzer" OFF)
find_package(nlohmann_json 3.12 REQUIRED)
find_package(Boost 1.85 CONFIG REQUIRED)
find_package(Threads REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS resources/action_catalog.json resources/dom_agent.js resources/legacy_value.js resources/key_layout.json resources/editing_commands.json)
file(READ resources/action_catalog.json RELAY_CATALOG)
file(READ resources/dom_agent.js RELAY_DOM)
file(READ resources/legacy_value.js RELAY_LEGACY_VALUE)
file(READ resources/key_layout.json RELAY_KEYS)
file(READ resources/editing_commands.json RELAY_EDITING)
configure_file(cmake/embedded_resources.hpp.in generated/embedded_resources.hpp @ONLY)
add_library(chromerelay src/io/cancellation.cpp src/mcp/request_queue.cpp src/io/json_contract.cpp src/io/path_access.cpp src/io/png_payload.cpp src/actions/action_catalog.cpp src/wire/devtools_channel.cpp src/mcp/stdio_endpoint.cpp src/browser/browser_workspace.cpp src/browser/frame_context.cpp src/browser/frame_geometry.cpp src/browser/input_observer.cpp src/browser/page_services.cpp src/browser/pointer_state.cpp src/actions/action_runtime.cpp src/actions/action_sequence.cpp src/actions/dom_actions.cpp src/actions/file_actions.cpp)
target_include_directories(chromerelay PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_compile_features(chromerelay PUBLIC cxx_std_20)
target_sources(chromerelay PRIVATE src/wire/devtools_message.cpp)
target_sources(chromerelay PRIVATE src/actions/keyboard_plan.cpp src/browser/keyboard_state.cpp)
target_sources(chromerelay PRIVATE src/actions/click_action.cpp)
target_link_libraries(chromerelay PUBLIC nlohmann_json::nlohmann_json PRIVATE Boost::headers Threads::Threads)
target_compile_options(chromerelay PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(CHROMERELAY_SANITIZE)
 target_compile_options(chromerelay PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(chromerelay PUBLIC -fsanitize=address,undefined)
endif()
add_executable(chrome-relay app/main.cpp)
target_link_libraries(chrome-relay PRIVATE chromerelay)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/ChromeRelayConfig.cmake.in
 ${CMAKE_CURRENT_BINARY_DIR}/ChromeRelayConfig.cmake
 INSTALL_DESTINATION lib/cmake/ChromeRelay)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/ChromeRelayConfigVersion.cmake
 COMPATIBILITY SameMajorVersion)
install(TARGETS chrome-relay chromerelay EXPORT ChromeRelayTargets
 RUNTIME DESTINATION bin ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(EXPORT ChromeRelayTargets NAMESPACE ChromeRelay:: DESTINATION lib/cmake/ChromeRelay)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/ChromeRelayConfig.cmake
 ${CMAKE_CURRENT_BINARY_DIR}/ChromeRelayConfigVersion.cmake DESTINATION lib/cmake/ChromeRelay)
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md dependencies.lock.json DESTINATION share/doc/ChromeRelay)
install(DIRECTORY docs licenses DESTINATION share/doc/ChromeRelay)
install(DIRECTORY examples DESTINATION share/ChromeRelay)
if(CHROMERELAY_BUILD_FUZZER)
 if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "Protocol fuzzing requires a Clang installation with libFuzzer")
 endif()
 add_executable(relay-protocol-fuzz tests/fuzz/protocol_fuzz.cpp src/io/json_contract.cpp src/io/cancellation.cpp src/actions/action_catalog.cpp src/mcp/stdio_endpoint.cpp src/wire/devtools_message.cpp)
 target_include_directories(relay-protocol-fuzz PRIVATE include ${CMAKE_CURRENT_BINARY_DIR}/generated)
 target_link_libraries(relay-protocol-fuzz PRIVATE nlohmann_json::nlohmann_json)
 target_compile_options(relay-protocol-fuzz PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
 target_link_options(relay-protocol-fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
